AWS BEDROCK — Converse API (works for ALL models) Python
import boto3

# Initialize Bedrock client (uses IAM role auth)
bedrock = boto3.client(service_name="bedrock-runtime", region_name="us-east-1")

# ── Basic chat completion ──────────────────────────────────
response = bedrock.converse(
    modelId="anthropic.claude-3-5-sonnet-20241022-v2:0",
    messages=[{
        "role": "user",
        "content": [{"text": "Explain RAG in 2 sentences"}]
    }],
    system=[{"text": "You are a helpful AI assistant."}],
    inferenceConfig={"maxTokens": 500, "temperature": 0.7}
)
text = response["output"]["message"]["content"][0]["text"]

# ── Streaming ──────────────────────────────────────────────
stream = bedrock.converse_stream(
    modelId="anthropic.claude-3-5-sonnet-20241022-v2:0",
    messages=[{"role": "user", "content": [{"text": "Tell me a story"}]}]
)
for event in stream["stream"]:
    if "contentBlockDelta" in event:
        print(event["contentBlockDelta"]["delta"]["text"], end="")

# ── Function calling on Bedrock ────────────────────────────
response = bedrock.converse(
    modelId="anthropic.claude-3-5-sonnet-20241022-v2:0",
    messages=[{"role": "user", "content": [{"text": "Get weather in Bengaluru"}]}],
    toolConfig={
        "tools": [{
            "toolSpec": {
                "name": "get_weather",
                "description": "Get weather for a city",
                "inputSchema": {"json": {
                    "type": "object",
                    "properties": {"city": {"type": "string"}},
                    "required": ["city"]
                }}
            }
        }]
    }
)

# ── Available models on Bedrock ────────────────────────────
# anthropic.claude-3-5-sonnet-20241022-v2:0  Claude 3.5 Sonnet (best)
# anthropic.claude-3-haiku-20240307-v1:0     Claude 3 Haiku (fast cheap)
# meta.llama3-70b-instruct-v1:0              Llama 3 70B
# amazon.titan-text-express-v1               Amazon Titan
# mistral.mistral-7b-instruct-v0:2           Mistral 7B
# cohere.command-r-plus-v1:0                 Cohere Command R+
💡

Bedrock key insight: The Converse API works identically across ALL model providers. Switch from Claude to Llama to Mistral by just changing modelId. No code changes needed.

AZURE OPENAI — Chat Completions Python
from openai import AzureOpenAI
import os

# Initialize Azure OpenAI client
client = AzureOpenAI(
    azure_endpoint=os.getenv("AZURE_OPENAI_ENDPOINT"),
    api_key=os.getenv("AZURE_OPENAI_KEY"),
    api_version="2024-02-01"
)

# ── Chat completion ────────────────────────────────────────
response = client.chat.completions.create(
    model="gpt-4o",          # This is your DEPLOYMENT NAME, not model name!
    messages=[
        {"role": "system", "content": "You are a helpful assistant."},
        {"role": "user",   "content": "Explain Azure AI Search"}
    ],
    temperature=0.7, max_tokens=500
)
print(response.choices[0].message.content)

# ── Streaming ──────────────────────────────────────────────
stream = client.chat.completions.create(
    model="gpt-4o",
    messages=[{"role": "user", "content": "Tell me a story"}],
    stream=True
)
for chunk in stream:
    if chunk.choices[0].delta.content:
        print(chunk.choices[0].delta.content, end="")

# ── Embeddings ──────────────────────────────────────────────
embed_resp = client.embeddings.create(
    model="text-embedding-3-small",  # your embedding deployment name
    input=["text to embed"]
)
vector = embed_resp.data[0].embedding  # 1536-dim list

# ── Azure AI Foundry SDK (new 2025) ─────────────────────────
from azure.ai.inference import ChatCompletionsClient
from azure.core.credentials import AzureKeyCredential

client2 = ChatCompletionsClient(
    endpoint="https://YOUR.services.ai.azure.com/models",
    credential=AzureKeyCredential(os.getenv("AZURE_KEY"))
)
# Supports GPT-4o, Llama, Mistral, Phi from ONE endpoint
💡

Azure key difference: The model param is your deployment name (not the OpenAI model name). You create deployments in Azure Portal and assign them custom names. Same OpenAI SDK — just use AzureOpenAI class instead of OpenAI.

GCP VERTEX AI — Gemini API Python
import vertexai
from vertexai.generative_models import GenerativeModel, Part, GenerationConfig

# Initialize (uses Application Default Credentials)
vertexai.init(project="your-gcp-project", location="us-central1")

# ── Basic Gemini call ──────────────────────────────────────
model = GenerativeModel("gemini-1.5-pro-002")
response = model.generate_content(
    "Explain RAG in 2 sentences",
    generation_config=GenerationConfig(temperature=0.7, max_output_tokens=500)
)
print(response.text)

# ── Multi-turn chat ────────────────────────────────────────
chat = model.start_chat()
r1 = chat.send_message("What is RAG?")
r2 = chat.send_message("Give me a Python example")  # keeps context!

# ── Streaming ──────────────────────────────────────────────
for chunk in model.generate_content("Tell me a story", stream=True):
    print(chunk.text, end="")

# ── Multimodal (text + image + video — unique to Gemini!) ──
with open("chart.png", "rb") as f:
    img = Part.from_data(data=f.read(), mime_type="image/png")
response = model.generate_content([img, "What does this chart show?"])

# ── 1M token context — feed entire documents! ──────────────
with open("huge_codebase.py", "r") as f:
    huge_doc = f.read()
response = model.generate_content(f"Find bugs in this code:\n{huge_doc}")
# Works for files up to 1M tokens — no chunking needed!

# ── Models available ───────────────────────────────────────
# gemini-1.5-pro-002    1M context, most capable, multimodal
# gemini-1.5-flash-002  fastest, cheapest, 1M context
# gemini-1.0-pro        stable widely-used model
💡

Gemini unique advantage: 1M token context window means you can feed entire codebases, hour-long video transcripts, or hundreds of documents in a single call — no chunking or RAG needed for many use cases.

MANAGED RAG

Managed RAG on Each Cloud

Knowledge Bases (AWS) · AI Search (Azure) · Vertex RAG Engine (GCP)

🟠 AWS — Knowledge Bases
  • Upload docs to S3 → auto-chunks + embeds
  • Default vector store: OpenSearch Serverless
  • Also supports: Pinecone, Redis, Aurora pgvector
  • RetrieveAndGenerate API — one call for full RAG
  • Retrieve API — retrieval only
  • Metadata filtering: filter by department, date etc
🔵 Azure — AI Search
  • Index schema with vector + content fields
  • Vectorizers auto-embed at index + query time
  • Semantic Ranker — Microsoft reranker built-in
  • Hybrid: keyword BM25 + vector in one request
  • Integrates natively with Azure OpenAI
  • Unique: Semantic Ranker is production-grade reranking
🟢 GCP — Vertex RAG Engine
  • Create RAG corpus → import from GCS
  • Auto-chunks + embeds documents
  • Pair with Gemini as a Tool
  • Vertex AI Search — Google-quality semantic search
  • Grounding with Google Search — cite live web
  • Unique: Google Search grounding for real-time facts

        "type": "KNOWLEDGE_BASE",

            "knowledgeBaseId": "YOUR_KB_ID",
            "modelArn": "arn:aws:bedrock:us-east-1::foundation-model/anthropic.claude-3-sonnet-20240229-v1:0",
            "retrievalConfiguration": {
                "vectorSearchConfiguration": {"numberOfResults": 5}

    }

# ── Retrieve only (no generation) ──────────────────────────

            "numberOfResults": 5,
            "filter": {  # metadata filtering
                "equals": {"key": "department", "value": "HR"}

        }
    }

AZURE AI SEARCH — Hybrid RAG Pipeline Python
from azure.search.documents import SearchClient
from azure.search.documents.models import VectorizedQuery
from azure.core.credentials import AzureKeyCredential
from openai import AzureOpenAI

search = SearchClient(
    endpoint="https://YOUR-SEARCH.search.windows.net",
    index_name="my-index",
    credential=AzureKeyCredential("YOUR_ADMIN_KEY")
)
aoai = AzureOpenAI(azure_endpoint="https://YOUR.openai.azure.com",
                   api_key="KEY", api_version="2024-02-01")

def embed(text):
    return aoai.embeddings.create(model="text-embedding-3-small", input=text).data[0].embedding

def hybrid_rag(query):
    # 1. Hybrid search: BM25 keyword + vector + Semantic Ranker
    results = search.search(
        search_text=query,                     # BM25 keyword
        vector_queries=[VectorizedQuery(      # dense vector
            vector=embed(query), k_nearest_neighbors=5, fields="contentVector"
        )],
        query_type="semantic",               # Semantic Ranker reranking
        semantic_configuration_name="my-config",
        select=["content", "title", "source"], top=5
    )
    context = "\n".join([r["content"] for r in results])
    # 2. Generate answer
    answer = aoai.chat.completions.create(
        model="gpt-4o",
        messages=[
            {"role": "system", "content": f"Answer from context:\n{context}"},
            {"role": "user",   "content": query}
        ], temperature=0
    )
    return answer.choices[0].message.content
GCP VERTEX AI RAG ENGINE Python
from vertexai.preview import rag
from vertexai.generative_models import GenerativeModel
from vertexai.preview.generative_models import Tool
import vertexai

vertexai.init(project="your-project", location="us-central1")

# ── Create corpus and import docs from GCS ─────────────────
corpus = rag.create_corpus(display_name="my-knowledge-base")
rag.import_files(corpus.name,
    paths=["gs://my-bucket/docs/"],
    chunk_size=512, chunk_overlap=100
)

# ── Retrieval only ─────────────────────────────────────────
results = rag.retrieval_query(
    rag_resources=[rag.RagResource(rag_corpus=corpus.name)],
    text="What is the refund policy?",
    similarity_top_k=5
)

# ── RAG + Gemini generation — attach as a Tool ─────────────
rag_tool = Tool.from_retrieval(
    retrieval=rag.Retrieval(source=rag.VertexRagStore(
        rag_resources=[rag.RagResource(rag_corpus=corpus.name)],
        similarity_top_k=5
    ))
)
model = GenerativeModel("gemini-1.5-pro-002", tools=[rag_tool])
response = model.generate_content("What is the refund policy?")
print(response.text)  # Gemini auto-retrieves and cites sources

# ── Grounding with Google Search (real-time facts) ──────────
from vertexai.generative_models import grounding
grounded_model = GenerativeModel(
    "gemini-1.5-pro-002",
    tools=[grounding.GoogleSearchRetrieval()]   # search the live web!
)
response = grounded_model.generate_content("What is the latest AI news?")
# Cites real web sources, no hallucination on current events

SAGEMAKER — Training Job + HPO Python
import sagemaker
from sagemaker.pytorch import PyTorch
from sagemaker.tuner import HyperparameterTuner, ContinuousParameter

role = sagemaker.get_execution_role()

# ── Define training job ────────────────────────────────────
estimator = PyTorch(
    entry_point="train.py",
    source_dir="./src",
    role=role,
    framework_version="2.1.0",
    py_version="py310",
    instance_type="ml.g5.xlarge",        # A10G GPU
    instance_count=1,
    hyperparameters={"epochs": 10, "lr": 0.001, "batch_size": 32},
    output_path="s3://my-bucket/output",
    use_spot_instances=True,             # up to 90% cost saving!
    max_wait=7200
)
estimator.fit({"train": "s3://my-bucket/train", "test": "s3://my-bucket/test"})

# ── Distributed multi-GPU training ────────────────────────
dist_estimator = PyTorch(
    entry_point="train.py",
    instance_type="ml.g5.12xlarge",    # 4 GPUs per node
    instance_count=4,                  # 16 GPUs total
    distribution={"torch_distributed": {"enabled": True}},
    ...
)

# ── Hyperparameter Tuning (HPO) ────────────────────────────
tuner = HyperparameterTuner(
    estimator=estimator,
    objective_metric_name="validation:loss",
    objective_type="Minimize",
    hyperparameter_ranges={
        "lr": ContinuousParameter(0.0001, 0.1),
    },
    max_jobs=20, max_parallel_jobs=4
)
tuner.fit({"train": "s3://my-bucket/train"})

SAGEMAKER PIPELINES — Full Conditional MLOps Pipeline Python
from sagemaker.workflow.pipeline import Pipeline
from sagemaker.workflow.steps import TrainingStep, ProcessingStep
from sagemaker.workflow.step_collections import RegisterModel
from sagemaker.workflow.conditions import ConditionGreaterThanOrEqualTo
from sagemaker.workflow.condition_step import ConditionStep
from sagemaker.workflow.functions import JsonGet

# Step 1: Data Processing
process_step = ProcessingStep(
    name="DataProcessing", processor=sklearn_processor,
    inputs=[...], outputs=[...], code="preprocess.py"
)

# Step 2: Training (depends on Step 1 output)
train_step = TrainingStep(
    name="ModelTraining", estimator=estimator,
    inputs={"train": process_step.properties.ProcessingOutputConfig.Outputs["train"].S3Output.S3Uri}
)

# Step 3: Evaluation
eval_step = ProcessingStep(
    name="ModelEvaluation", processor=eval_processor,
    inputs=[ProcessingInput(source=train_step.properties.ModelArtifacts.S3ModelArtifacts, ...)],
    code="evaluate.py", property_files=[evaluation_report]
)

# Step 4: Conditional Register — only if accuracy >= 85%
condition = ConditionStep(
    name="AccuracyGate",
    conditions=[ConditionGreaterThanOrEqualTo(
        left=JsonGet(step_name=eval_step.name, property_file=evaluation_report, json_path="metrics.accuracy"),
        right=0.85
    )],
    if_steps=[RegisterModel(
        name="RegisterModel", estimator=estimator,
        model_data=train_step.properties.ModelArtifacts.S3ModelArtifacts,
        model_package_group_name="MyModelGroup",
        approval_status="PendingManualApproval"
    )],
    else_steps=[]
)

# Build and run
pipeline = Pipeline(
    name="MyMLPipeline",
    steps=[process_step, train_step, eval_step, condition]
)
pipeline.upsert(role_arn=role)
execution = pipeline.start()
execution.wait()  # blocks until done

Detect data drift, performance degradation, and operational issues

What to Monitor — All Clouds
📊Data Drift

Input feature distribution shifts from training baseline. Model sees data it was never trained on.

Signals

  • KS test / PSI score for numerical features
  • Chi-squared for categorical features
  • Covariate shift — X distribution changed
  • Concept drift — P(Y|X) changed
📉Model Performance Drift

Accuracy, F1, or business KPIs degrade in production vs training baseline.

Approaches

  • Ground truth comparison (delayed labels)
  • Prediction confidence tracking
  • Shadow deployment — compare models
  • Proxy metrics when labels unavailable
⚡Operational Metrics

Latency, throughput, error rates, cost per inference for real-time endpoints.

Track

  • P50/P95/P99 latency per endpoint
  • Invocation error rate — 4xx, 5xx
  • Throughput — requests/second
  • Token cost per request (LLMs)

# ── LLM-as-judge pattern ──────────────────────────────────
def llm_judge(question, answer, expected, client):
    prompt = f"""Score this answer 0-10.
Question: {question}
Expected: {expected}
Actual: {answer}
Return ONLY JSON: {{"score": N, "reason": "brief explanation"}}"""

    result = client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": prompt}],
        response_format={"type": "json_object"}
    )
    import json
    return json.loads(result.choices[0].message.content)

# → {"score": 9, "reason": "Correct and concise"}

MODEL ROUTING + SEMANTIC CACHE — Smart Cost Reduction Python
# ── Model routing: cheap for simple, expensive for hard ───
def smart_route(query: str, client) -> str:
    # Classify complexity first with cheapest model
    classify = client.chat.completions.create(
        model="gpt-4o-mini",          # ~100× cheaper than gpt-4o
        messages=[{"role": "user",
            "content": f"Is this query simple (yes/no)?\nQuery: {query}"}],
        max_tokens=5
    )
    is_simple = "yes" in classify.choices[0].message.content.lower()
    model = "gpt-4o-mini" if is_simple else "gpt-4o"
    
    response = client.chat.completions.create(
        model=model,
        messages=[{"role": "user", "content": query}]
    )
    return response.choices[0].message.content

    name="my-guardrail",
    description="Prevent off-topic and PII leakage",
    topicPolicyConfig={

            "name": "OffTopicPolitics",
            "definition": "Questions about political topics",

        ]
